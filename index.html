<script src="https://cdn.jsdelivr.net/npm/face-api.js@0.22.2/dist/face-api.min.js"></script>
<script>
(() => {
  const MODEL_URL = 'https://cdn.jsdelivr.net/gh/justadudewhohacks/face-api.js@master/weights';
  const video = document.getElementById('video');
  const placeholder = document.getElementById('cameraPlaceholder');
  const statusEl = document.getElementById('modelStatus');
  const resultEl = document.getElementById('result');
  const registerBtn = document.getElementById('registerBtn');
  const startBtn = document.getElementById('startBtn');
  const scanBtn = document.getElementById('scanBtn');
  const stopBtn = document.getElementById('stopBtn');
  let modelsReady = false, stream = null, references = [], busy = false;

  function setResult(main, detail, cls='') {
    resultEl.className = cls;
    resultEl.innerHTML = '';
    const strong = document.createElement('span');
    strong.textContent = main;
    const small = document.createElement('small');
    small.textContent = detail;
    resultEl.append(strong, small);
  }
  function updateButtons() {
    registerBtn.disabled = !modelsReady;
    startBtn.disabled = !modelsReady;
    scanBtn.disabled = !stream || references.length === 0 || busy;
    stopBtn.disabled = !stream;
  }

  async function loadModels() {
    try {
      if (!window.faceapi) throw new Error('The face recognition library could not load. Check your internet connection and reload.');
      await Promise.all([
        faceapi.nets.tinyFaceDetector.loadFromUri(MODEL_URL),
        faceapi.nets.faceLandmark68Net.loadFromUri(MODEL_URL),
        faceapi.nets.faceRecognitionNet.loadFromUri(MODEL_URL)
      ]);
      modelsReady = true;
      statusEl.textContent = 'Models loaded. You can now register photos and start the camera.';
      statusEl.style.color = 'var(--green)';
      updateButtons();
    } catch (err) {
      statusEl.textContent = 'Could not load models: ' + err.message + ' Please check internet and reload this page.';
      statusEl.style.color = 'var(--red)';
    }
  }

  async function descriptorFromImage(file) {
    const url = URL.createObjectURL(file);
    const img = new Image();
    try {
      await new Promise((resolve, reject) => {
        img.onload = resolve;
        img.onerror = () => reject(new Error('This image could not be opened.'));
        img.src = url;
      });
      const detection = await faceapi.detectSingleFace(img, new faceapi.TinyFaceDetectorOptions({inputSize: 320, scoreThreshold: 0.35}))
        .withFaceLandmarks().withFaceDescriptor();
      if (!detection) throw new Error('No clear face found. Try a front-facing photo with good lighting.');
      return detection.descriptor;
    } finally {
      URL.revokeObjectURL(url);
    }
  }

  registerBtn.addEventListener('click', async () => {
    if (busy) return;
    busy = true; updateButtons();
    const next = [];
    try {
      for (let i = 0; i < 5; i++) {
        const file = document.getElementById('photo' + i).files[0];
        if (!file) continue;
        const label = document.getElementById('name' + i).value.trim() || ('Reference ' + (i + 1));
        statusEl.textContent = 'Processing ' + label + '…';
        const descriptor = await descriptorFromImage(file);
        next.push({label, descriptor});
      }
      if (!next.length) {
        setResult('NO PHOTOS SELECTED', 'Choose at least one reference photo above.');
      } else {
        references = next;
        statusEl.textContent = references.length + ' reference photo(s) registered in this browser session.';
        statusEl.style.color = 'var(--green)';
        setResult('REFERENCES READY', references.length + ' reference photo(s) registered. Start the camera and scan.', 'result-match');
      }
    } catch (err) {
      references = [];
      setResult('REGISTRATION FAILED', err.message, 'result-warn');
      statusEl.textContent = 'Please correct the photo and try registering again.';
      statusEl.style.color = 'var(--amber)';
    } finally {
      busy = false; updateButtons();
    }
  });

  document.getElementById('clearBtn').addEventListener('click', () => {
    references = [];
    for (let i = 0; i < 5; i++) {
      document.getElementById('photo' + i).value = '';
      document.getElementById('name' + i).value = '';
    }
    setResult('REFERENCES CLEARED', 'Add reference photos to use the demo.');
    updateButtons();
  });

  startBtn.addEventListener('click', async () => {
    try {
      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        throw new Error('Camera access is unavailable. Open this website using its HTTPS link in Chrome.');
      }
      if (stream) stream.getTracks().forEach(t => t.stop());
      stream = await navigator.mediaDevices.getUserMedia({
        audio: false,
        video: {facingMode: 'user', width: {ideal: 640}, height: {ideal: 480}}
      });
      video.srcObject = stream;
      video.hidden = false;
      placeholder.hidden = true;
      await new Promise((resolve) => {
        if (video.readyState >= 1) return resolve();
        video.onloadedmetadata = () => resolve();
      });
      await video.play();
      setResult('CAMERA READY', 'Tap “Scan face” when a face is visible.');
    } catch (err) {
      stream = null;
      video.hidden = true; placeholder.hidden = false;
      setResult('CAMERA UNAVAILABLE', err.message, 'result-warn');
    }
    updateButtons();
  });

  scanBtn.addEventListener('click', async () => {
    if (!stream || !references.length || busy) return;
    busy = true; updateButtons();
    try {
      if (video.readyState < 2 || !video.videoWidth) throw new Error('Camera is still starting. Wait a moment and try again.');
      const canvas = document.createElement('canvas');
      canvas.width = video.videoWidth; canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      ctx.translate(canvas.width, 0); ctx.scale(-1, 1);
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      const detections = await faceapi.detectAllFaces(canvas, new faceapi.TinyFaceDetectorOptions({inputSize: 320, scoreThreshold: 0.35}))
        .withFaceLandmarks().withFaceDescriptors();
      if (!detections.length) {
        setResult('NO FACE DETECTED', 'Move into the camera view, improve lighting and scan again.', 'result-warn');
      } else if (detections.length > 1) {
        setResult('ONE PERSON AT A TIME', 'Please make sure only one face is visible for this demonstration.', 'result-warn');
      } else {
        let best = {distance: Infinity, label: ''};
        for (const ref of references) {
          const distance = faceapi.euclideanDistance(detections[0].descriptor, ref.descriptor);
          if (distance < best.distance) best = {distance, label: ref.label};
        }
        if (best.distance < 0.48) {
          setResult('STUDENT — POSSIBLE REGISTERED MATCH', 'Closest reference: ' + best.label + '. Confirm identity separately.', 'result-match');
        } else if (best.distance < 0.58) {
          setResult('UNCERTAIN — CHECK MANUALLY', 'The result is not reliable enough to suggest a match. Confirm identity separately.', 'result-warn');
        } else {
          setResult('NOT VERIFIED — NO MATCH', 'No close match in the registered photos. This does not prove the person is a stranger.', 'result-none');
        }
      }
    } catch (err) {
      setResult('SCAN ERROR', err.message, 'result-warn');
    } finally {
      busy = false; updateButtons();
    }
  });

  stopBtn.addEventListener('click', () => {
    if (stream) stream.getTracks().forEach(t => t.stop());
    stream = null;
    video.srcObject = null;
    video.hidden = true; placeholder.hidden = false;
    setResult('CAMERA STOPPED', 'Start the camera when you are ready.');
    updateButtons();
  });

  window.addEventListener('beforeunload', () => {
    if (stream) stream.getTracks().forEach(t => t.stop());
  });
  loadModels();
})();
</script>
